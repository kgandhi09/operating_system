cmake_minimum_required(VERSION 3.20)
project(jk_nodeinfra VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
option(JK_BUILD_EXAMPLES "Build local publisher/subscriber examples" ON)
set(JKBUF_HOST_EXECUTABLE "" CACHE FILEPATH "Native jkbuf compiler used when cross-compiling")
add_subdirectory(idl)
add_subdirectory(runtime)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(BuildJkbuf)
if(NOT CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  set(CMAKE_MODULE_PATH "${CMAKE_MODULE_PATH}" PARENT_SCOPE)
endif()
if(JK_BUILD_EXAMPLES OR BUILD_TESTING)
  add_subdirectory(examples)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
