if(CMAKE_CROSSCOMPILING)
  if(NOT JKBUF_HOST_EXECUTABLE)
    message(FATAL_ERROR "Cross builds require -DJKBUF_HOST_EXECUTABLE=/path/to/native/jkbuf (or disable examples and tests)")
  endif()
  set(schema_compiler ${JKBUF_HOST_EXECUTABLE})
else()
  set(schema_compiler $<TARGET_FILE:jkbuf>)
endif()
set(example_header ${CMAKE_CURRENT_BINARY_DIR}/sensor.h)
add_custom_command(OUTPUT ${example_header}
  COMMAND ${schema_compiler} -o ${example_header} ${CMAKE_CURRENT_SOURCE_DIR}/sensor.ridl
  DEPENDS jkbuf ${CMAKE_CURRENT_SOURCE_DIR}/sensor.ridl VERBATIM)
add_custom_target(example_schema DEPENDS ${example_header})
foreach(app publisher subscriber)
  add_executable(jk_${app} ${app}.cpp)
  add_dependencies(jk_${app} example_schema)
  target_include_directories(jk_${app} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(jk_${app} PRIVATE jk_nodeinfra)
endforeach()
